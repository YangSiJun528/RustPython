import locale
import json

results = []
for name in ('C', 'en_US.UTF-8', 'de_DE.UTF-8', 'fr_FR.UTF-8', 'hi_IN.UTF-8'):
    try:
        effective = locale.setlocale(locale.LC_ALL, name)
    except locale.Error as exc:
        results.append({'locale': name, 'unavailable': str(exc)})
        continue
    conv = locale.localeconv()
    results.append({'locale': name, 'effective': effective, 'conv': {k: conv[k] for k in ('decimal_point', 'thousands_sep', 'grouping')}})
    for value in (0, 1234, -123456789, 123456789012345678901234567890, 1234.5, -12345.25, 1.25e12):
        specs = ('n', '+n', ' n', '20n', '>20n', '*<20n', '^20n', '020n', '+020n', '0=20n')
        if isinstance(value, float):
            specs += ('.10n', '20.10n', '020.10n')
        for spec in specs:
            try:
                result = format(value, spec)
            except Exception as exc:
                result = [type(exc).__name__, str(exc)]
            results.append({'locale': name, 'value': repr(value), 'spec': spec, 'result': result})
print(json.dumps(results, ensure_ascii=True, indent=2))
