for width in [-5, -3, -1, 0, 1, 3, 5]:
    for precision in [-3, -1, 0, 1, 3, 5]:
        print('str', width, precision, repr('%*.*s' % (width, precision, 'abc')))
        print('bytes', width, precision, repr(b'%*.*s' % (width, precision, b'abc')))
for fmt in ['%*s', '%-*s', '%0*d', '%+*d', '% *d']:
    value = 'abc' if fmt.endswith('s') else 12
    print('flags', fmt, repr(fmt % (-5, value)))
