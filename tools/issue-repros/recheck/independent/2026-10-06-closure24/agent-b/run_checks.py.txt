def pair(issue,label,filename,args=(),flags=(),timeout=45,env=None,rationale='Compare original or explicitly labeled expanded behavior with CPython; assertion failures, warnings, exceptions and counts require interpretation.'):
    path=str(ROOT/'inputs'/filename) if not filename.startswith('/') else filename
    for kind,exe in [('rp',BIN),('cp',CP)]:
        record=run(f'{issue}-{label}-{kind}',[exe,'-B',*flags,path,*args],issue,timeout,dict(ENV,**(env or {})),path,rationale)
        print(pathlib.Path(record['stdout_file']).read_text()[:1500],flush=True)
        print(pathlib.Path(record['stderr_file']).read_text()[:800],flush=True)
